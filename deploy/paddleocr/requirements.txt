# Only the FastAPI server layer. paddlepaddle and paddleocr come from the
# paddlecloud base image (Python 3.7), so these pins target 3.7 compatibility:
# FastAPI 0.95.x is the last line that supports Python 3.7 before 0.99 dropped
# it, and uvicorn 0.22 is the matching server.
#
# python-multipart is intentionally NOT installed. server.py only uses JSON
# request bodies (pydantic BaseModel) — no Form, File, or UploadFile imports —
# so the multipart parser is dead weight. Dropping it eliminates three CVEs
# (GHSA-2jv5-9r88-3w3p, GHSA-59g5-xgcq-4qw3, GHSA-wp53-j4wj-2cfg) whose
# fixes all require Python 3.8+ or 3.10+, which would force a base image
# rebase. If a future endpoint ever needs file uploads, re-add it then —
# but use a version that's actually compatible with the runtime Python.
fastapi==0.95.2
uvicorn==0.22.0
